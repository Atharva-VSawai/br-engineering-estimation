    67→function checkCompleteness(config: ProjectConfig): SectionCheck[] {
    68→  const ioTotal =
    69→    config.io.digitalInputs + config.io.digitalOutputs +
    70→    config.io.analogInputs + config.io.analogOutputs +
    71→    config.io.safetyIO + config.io.encoderCounterModules +
    72→    config.io.temperatureModules + config.io.communicationIO +
    73→    config.io.specialModules;
    74→
    75→  const motionFeatures = [
    76→    config.motion.homingRequired, config.motion.positioning,
    77→    config.motion.velocityControl, config.motion.torqueControl,
    78→    config.motion.synchronization, config.motion.masterSlave,
    79→    config.motion.electronicGearing, config.motion.electronicCamming,
    80→    config.motion.coordinatedMotion, config.motion.interpolation,
    81→    config.motion.complexMotionProfiles, config.motion.axisDiagnostics,
    82→  ].filter(Boolean).length;
    83→
    84→  const hmiFeatures = [
    85→    config.hmi.alarmManagement, config.hmi.recipeManagement,
    86→    config.hmi.trendVisualization, config.hmi.userManagement,
    87→    config.hmi.machineDiagnostics, config.hmi.manualMode,
    88→    config.hmi.automaticMode, config.hmi.maintenanceScreens,
    89→    config.hmi.parameterManagement,
    90→  ].filter(Boolean).length;
    91→
    92→  const commActive = config.communication.protocols.some((p) => p.enabled);
    93→  const commIntegrations = [
    94→    config.communication.plcToPlc, config.communication.mesIntegration,
    95→    config.communication.scadaIntegration, config.communication.cloudIIoTIntegration,
    96→  ].filter(Boolean).length;
    97→
    98→  const iiotFeatures = [
    99→    config.iiot.ipcRequired, config.iiot.iiotRequired,
   100→    config.iiot.iiotConnector, config.iiot.iiotServices,
   101→    config.iiot.iiotEdgeDevice, config.iiot.cloudConnectivity,
   102→    config.iiot.machineDataCollection, config.iiot.remoteMaintenance,
   103→    config.iiot.opcUa, config.iiot.dataLogging, config.iiot.analyticsIntegration,
   104→  ].filter(Boolean).length;
   105→
   106→  return [
   107→    {
   108→      label: 'Project',
   109→      configured: !!(config.project.name && config.project.name.trim()),
   110→    },
   111→    {
   112→      label: 'Controller',
   113→      configured: config.controller.quantity > 1 || config.controller.performance !== 'Standard' || !!config.controller.communicationInterfaces,
   114→    },
   115→    {
   116→      label: 'I/O',
   117→      configured: ioTotal > 0,
   118→    },
   119→    {
   120→      label: 'Motion',
   121→      configured: config.motion.totalAxes > 0 || motionFeatures > 0,
   122→    },
   123→    {
   124→      label: 'HMI',
   125→      configured: config.hmi.screens > 0 || hmiFeatures > 0,
   126→    },
   127→    {
   128→      label: 'Vision',
   129→      configured: config.vision.enabled,
   130→    },
   131→    {
   132→      label: 'Safety',
   133→      configured: config.safety.enabled,
   134→    },
   135→    {
   136→      label: 'Communication',
   137→      configured: commActive || commIntegrations > 0,
   138→    },
   139→    {
   140→      label: 'Mechatronics',
   141→      configured: config.mechatronics.type !== 'None' && config.mechatronics.type !== '',
   142→    },
   143→    {
   144→      label: 'Robotics',
   145→      configured: config.robotics.enabled,
   146→    },
   147→    {
   148→      label: 'IIoT',
   149→      configured: iiotFeatures > 0,
   150→    },
   151→  ];
   152→}
