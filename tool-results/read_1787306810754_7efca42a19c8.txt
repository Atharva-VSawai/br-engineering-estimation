    92→function checkCompleteness(config: ProjectConfig): SectionCheck[] {
    93→  const ioTotal = config.io.digitalInputs + config.io.digitalOutputs + config.io.analogInputs + config.io.analogOutputs + config.io.safetyIO + config.io.encoderCounterModules + config.io.temperatureModules + config.io.communicationIO + config.io.specialModules;
    94→  const motionFeatures = [config.motion.homingRequired, config.motion.positioning, config.motion.velocityControl, config.motion.torqueControl, config.motion.synchronization, config.motion.masterSlave, config.motion.electronicGearing, config.motion.electronicCamming, config.motion.coordinatedMotion, config.motion.interpolation, config.motion.complexMotionProfiles, config.motion.axisDiagnostics].filter(Boolean).length;
    95→  const hmiFeatures = [config.hmi.alarmManagement, config.hmi.recipeManagement, config.hmi.trendVisualization, config.hmi.userManagement, config.hmi.machineDiagnostics, config.hmi.manualMode, config.hmi.automaticMode, config.hmi.maintenanceScreens, config.hmi.parameterManagement].filter(Boolean).length;
    96→  const commActive = config.communication.protocols.some((p) => p.enabled);
    97→  const commIntegrations = [config.communication.plcToPlc, config.communication.mesIntegration, config.communication.scadaIntegration, config.communication.cloudIIoTIntegration].filter(Boolean).length;
    98→  const iiotFeatures = [config.iiot.ipcRequired, config.iiot.iiotRequired, config.iiot.iiotConnector, config.iiot.iiotServices, config.iiot.iiotEdgeDevice, config.iiot.cloudConnectivity, config.iiot.machineDataCollection, config.iiot.remoteMaintenance, config.iiot.opcUa, config.iiot.dataLogging, config.iiot.analyticsIntegration].filter(Boolean).length;
    99→  return [
   100→    { label: 'Project', configured: !!(config.project.name && config.project.name.trim()) },
   101→    { label: 'Controller', configured: config.controller.quantity > 1 || config.controller.performance !== 'Standard' || !!config.controller.communicationInterfaces },
   102→    { label: 'I/O', configured: ioTotal > 0 },
   103→    { label: 'Motion', configured: config.motion.totalAxes > 0 || motionFeatures > 0 },
   104→    { label: 'HMI', configured: config.hmi.screens > 0 || hmiFeatures > 0 },
   105→    { label: 'Vision', configured: config.vision.enabled },
   106→    { label: 'Safety', configured: config.safety.enabled },
   107→    { label: 'Communication', configured: commActive || commIntegrations > 0 },
   108→    { label: 'Mechatronics', configured: config.mechatronics.type !== 'None' && config.mechatronics.type !== '' },
   109→    { label: 'Robotics', configured: config.robotics.enabled },
   110→    { label: 'IIoT', configured: iiotFeatures > 0 },
   111→  ];
   112→}

   114→function getEffortDrivers(c: ProjectConfig): string[] {
   115→  const drivers: string[] = [];
   116→  if (c.motion.totalAxes > 0) drivers.push(`${c.motion.totalAxes} motion axes`);
   117→  if (c.motion.electronicCamming) drivers.push('Electronic camming');
   118→  if (c.motion.coordinatedMotion) drivers.push('Coordinated motion');
   119→  if (c.motion.interpolation) drivers.push('Multi-axis interpolation');
   120→  if (c.motion.complexMotionProfiles) drivers.push('Complex motion profiles');
   121→  if (c.mechatronics.type !== 'None') drivers.push(`${c.mechatronics.type} mechatronics`);
   122→  if (c.robotics.enabled) drivers.push(`${c.robotics.quantity} robot(s)`);
   123→  if (c.vision.enabled) drivers.push(`${c.vision.cameras} camera(s)`);
   124→  if (c.safety.enabled) drivers.push(`Safety system (${c.safety.safetyIOCount} I/O)`);
   125→  if (c.hmi.screens > 0) drivers.push(`${c.hmi.screens} HMI screen(s)`);
   126→  const enabledProtocols = c.communication.protocols.filter((p) => p.enabled);
   127→  if (enabledProtocols.length > 0) drivers.push(`${enabledProtocols.length} communication protocol(s)`);
   128→  if (c.communication.mesIntegration) drivers.push('MES integration');
   129→  if (c.communication.scadaIntegration) drivers.push('SCADA integration');
   130→  if (c.iiot.ipcRequired) drivers.push(`IPC (${c.iiot.ipcModel})`);
   131→  if (c.iiot.cloudConnectivity) drivers.push('Cloud/IIoT connectivity');
   132→  const totalIO = c.io.digitalInputs + c.io.digitalOutputs + c.io.analogInputs + c.io.analogOutputs;
   133→  if (totalIO > 100) drivers.push(`${totalIO} I/O points`);
   134→  if (c.project.machineStations > 1) drivers.push(`${c.project.machineStations} machine stations`);
   135→  if (c.project.projectVariants > 1) drivers.push(`${c.project.projectVariants} product variants`);
   136→  return drivers;
   137→}
