    54→function checkCompleteness(config: ProjectConfig): { label: string; configured: boolean }[] {
    55→  const c = config;
    56→  const ioTotal = c.io.digitalInputs + c.io.digitalOutputs + c.io.analogInputs + c.io.analogOutputs + c.io.safetyIO + c.io.encoderCounterModules + c.io.temperatureModules + c.io.communicationIO + c.io.specialModules;
    57→  const motionFeatures = [c.motion.homingRequired, c.motion.positioning, c.motion.velocityControl, c.motion.torqueControl, c.motion.synchronization, c.motion.masterSlave, c.motion.electronicGearing, c.motion.electronicCamming, c.motion.coordinatedMotion, c.motion.interpolation, c.motion.complexMotionProfiles, c.motion.axisDiagnostics].filter(Boolean).length;
    58→  const hmiFeatures = [c.hmi.alarmManagement, c.hmi.recipeManagement, c.hmi.trendVisualization, c.hmi.userManagement, c.hmi.machineDiagnostics, c.hmi.manualMode, c.hmi.automaticMode, c.hmi.maintenanceScreens, c.hmi.parameterManagement].filter(Boolean).length;
    59→  const commActive = c.communication.protocols.some((p) => p.enabled);
    60→  const commIntegrations = [c.communication.plcToPlc, c.communication.mesIntegration, c.communication.scadaIntegration, c.communication.cloudIIoTIntegration].filter(Boolean).length;
    61→  const iiotFeatures = [c.iiot.ipcRequired, c.iiot.iiotRequired, c.iiot.iiotConnector, c.iiot.iiotServices, c.iiot.iiotEdgeDevice, c.iiot.cloudConnectivity, c.iiot.machineDataCollection, c.iiot.remoteMaintenance, c.iiot.opcUa, c.iiot.dataLogging, c.iiot.analyticsIntegration].filter(Boolean).length;
    62→  return [
    63→    { label: 'Project', configured: !!(c.project.name && c.project.name.trim()) },
    64→    { label: 'Controller', configured: c.controller.quantity > 1 || c.controller.performance !== 'Standard' || !!c.controller.communicationInterfaces },
    65→    { label: 'I/O', configured: ioTotal > 0 },
    66→    { label: 'Motion', configured: c.motion.totalAxes > 0 || motionFeatures > 0 },
    67→    { label: 'HMI', configured: c.hmi.screens > 0 || hmiFeatures > 0 },
    68→    { label: 'Vision', configured: c.vision.enabled },
    69→    { label: 'Safety', configured: c.safety.enabled },
    70→    { label: 'Communication', configured: commActive || commIntegrations > 0 },
    71→    { label: 'Mechatronics', configured: c.mechatronics.type !== 'None' && c.mechatronics.type !== '' },
    72→    { label: 'Robotics', configured: c.robotics.enabled },
    73→    { label: 'IIoT', configured: iiotFeatures > 0 },
    74→  ];
    75→}

    91→// ── Donut Chart ───────────────────────────────────────────────────────────────
    92→
    93→function drawDonutChart(
    94→  doc: jsPDF,
    95→  cx: number,
    96→  cy: number,
    97→  outerR: number,
    98→  innerR: number,
    99→  data: { label: string; value: number }[],
   100→  totalHours: number,
   101→): void {
   102→  let startAngle = -Math.PI / 2;
   103→
   104→  for (let i = 0; i < data.length; i++) {
   105→    const item = data[i];
   106→    if (item.value <= 0) continue;
   107→    const sliceAngle = (item.value / totalHours) * 2 * Math.PI;
   108→    const endAngle = startAngle + sliceAngle;
   109→    const color = DONUT_COLORS[i % DONUT_COLORS.length];
   110→
   111→    doc.setFillColor(color[0], color[1], color[2]);
   112→    doc.setGState(new (doc as any).GState({ opacity: 0.9 }));
   113→
   114→    // Draw filled arc segments using small triangles for approximation
   115→    const steps = Math.max(Math.ceil(sliceAngle / 0.05), 8);
   116→    // Outer arc
   117→    for (let s = 0; s <= steps; s++) {
   118→      const angle = startAngle + (sliceAngle * s) / steps;
   119→      const px = cx + outerR * Math.cos(angle);
   120→      const py = cy + outerR * Math.sin(angle);
   121→      if (s === 0) doc.moveTo(px, py);
   122→      else doc.lineTo(px, py);
   123→    }
   124→    // Inner arc (reverse)
   125→    for (let s = steps; s >= 0; s--) {
   126→      const angle = startAngle + (sliceAngle * s) / steps;
   127→      const px = cx + innerR * Math.cos(angle);
   128→      const py = cy + innerR * Math.sin(angle);
   129→      doc.lineTo(px, py);
   130→    }
   131→    doc.fill();
   132→
   133→    doc.setGState(new (doc as any).GState({ opacity: 1 }));
   134→
   135→    startAngle = endAngle;
   136→  }
   137→
   138→  // Center text
   139→  doc.setFont('helvetica', 'bold');
   140→  doc.setFontSize(16);
   141→  doc.setTextColor(50, 50, 50);
   142→  doc.text(`${Math.round(totalHours)}h`, cx, cy + 1, { align: 'center', baseline: 'middle' });
   143→  doc.setFont('helvetica', 'normal');
   144→  doc.setFontSize(7);
   145→  doc.setTextColor(120, 120, 120);
   146→  doc.text('Total Effort', cx, cy + 7, { align: 'center', baseline: 'middle' });
   147→
   148→  // Legend (to the right of the donut)
   149→  const legendX = cx + outerR + 10;
   150→  let legendY = cy - (data.length * 5) / 2;
   151→
   152→  for (let i = 0; i < data.length; i++) {
   153→    if (data[i].value <= 0) continue;
   154→    const color = DONUT_COLORS[i % DONUT_COLORS.length];
   155→    doc.setFillColor(color[0], color[1], color[2]);
   156→    doc.roundedRect(legendX, legendY - 1.5, 3, 3, 0.5, 0.5, 'F');
   157→
   158→    doc.setFont('helvetica', 'normal');
   159→    doc.setFontSize(6.5);
   160→    doc.setTextColor(60, 60, 60);
   161→    const pct = totalHours > 0 ? Math.round((data[i].value / totalHours) * 100) : 0;
   162→    doc.text(`${data[i].label} (${pct}%)`, legendX + 5, legendY, { align: 'left', baseline: 'middle' });
   163→    legendY += 5;
   164→  }
   165→}

   167→// ── Horizontal Bar Chart ─────────────────────────────────────────────────────
   168→
   169→function drawHorizontalBarChart(
   170→  doc: jsPDF,
   171→  x: number,
   172→  y: number,
   173→  width: number,
   174→  barHeight: number,
   175→  gap: number,
   176→  items: { label: string; value: number; max: number; color: [number, number, number]; valueLabel?: string }[],
   177→  showTotal?: { label: string; value: number; color: [number, number, number] },
   178→): number {
   179→  const labelWidth = 42;
   180→  const barStartX = x + labelWidth + 4;
   181→  const barMaxWidth = width - labelWidth - 30;
   182→  const maxVal = Math.max(...items.map((it) => it.max), 1);
   183→
   184→  let curY = y;
   185→
   186→  for (const item of items) {
   187→    doc.setFont('helvetica', 'normal');
   188→    doc.setFontSize(7);
   189→    doc.setTextColor(50, 50, 50);
   190→    doc.text(item.label, x, curY + barHeight / 2 + 1, { align: 'right', baseline: 'middle' });
   191→
   192→    const barW = Math.max((item.value / maxVal) * barMaxWidth, 2);
   193→    doc.setFillColor(...item.color);
   194→    doc.roundedRect(barStartX, curY, barW, barHeight, 1.5, 1.5, 'F');
   195→
   196→    doc.setFont('helvetica', 'bold');
   197→    doc.setFontSize(7);
   198→    doc.setTextColor(50, 50, 50);
   199→    const vLabel = item.valueLabel ?? `${item.value.toFixed(1)}h`;
   200→    doc.text(vLabel, barStartX + barW + 3, curY + barHeight / 2 + 1, { align: 'left', baseline: 'middle' });
   201→
   202→    curY += barHeight + gap;
   203→  }
   204→
   205→  if (showTotal) {
   206→    curY -= gap / 2;
   207→    doc.setDrawColor(220, 220, 220);
   208→    doc.setLineWidth(0.3);
   209→    doc.line(barStartX, curY, barStartX + barMaxWidth, curY);
   210→    curY += gap / 2;
   211→
   212→    doc.setFont('helvetica', 'bold');
   213→    doc.setFontSize(7);
   214→    doc.setTextColor(...BR_ORANGE);
   215→    doc.text(showTotal.label, x, curY + barHeight / 2 + 1, { align: 'right', baseline: 'middle' });
   216→
   217→    const barW = Math.max((showTotal.value / maxVal) * barMaxWidth, 2);
   218→    doc.setFillColor(...showTotal.color);
   219→    doc.roundedRect(barStartX, curY, barW, barHeight, 1.5, 1.5, 'F');
   220→
   221→    doc.text(`${showTotal.value.toFixed(1)}h`, barStartX + barW + 3, curY + barHeight / 2 + 1, { align: 'left', baseline: 'middle' });
   222→    curY += barHeight + gap;
   223→  }
   224→
   225→  return curY;
   226→}

   230→function drawComplexityProfile(
   231→  doc: jsPDF,
   232→  x: number,
   233→  y: number,
   234→  width: number,
   235→  barHeight: number,
   236→  gap: number,
   237→  config: ProjectConfig,

   240→  const barStartX = x + labelWidth + 4;
   241→  const barMaxWidth = width - labelWidth - 25;
   242→
   243→  let curY = y;
   244→
   245→  for (const dim of DIMENSIONS) {
   246→    const level = config.complexity[dim.key];
   247→    const val = COMPLEXITY_VALUE[level] || 0.25;

   250→    doc.setFont('helvetica', 'normal');

   254→

   256→    doc.setFillColor(color[0], color[1], color[2]);
   257→    doc.setGState(new (doc as any).GState({ opacity: 0.85 }));
   258→    doc.roundedRect(barStartX, curY, barW, barHeight, 1.5, 1.5, 'F');
   259→    doc.setGState(new (doc as any).GState({ opacity: 1 }));
   260→
   261→    doc.setFont('helvetica', 'bold');
   262→    doc.setFontSize(7);

   265→
   266→    curY += barHeight + gap;
   267→  }
   268→
   269→  return curY;
   270→}

   272→// ── Timeline / Gantt Chart ────────────────────────────────────────────────────
   273→
   274→function drawTimelineChart(
   275→  doc: jsPDF,
   276→  x: number,
   277→  y: number,
   278→  width: number,
   279→  barHeight: number,
   280→  gap: number,
   281→  phases: { phase: string; weeks: number }[],
   282→  totalWeeks: number,
   283→): number {
   284→  const labelWidth = 45;
   285→  const barStartX = x + labelWidth + 4;
   286→  const barMaxWidth = width - labelWidth - 28;
   287→  const colors: [number, number, number][] = [
   288→    [59, 130, 246],
   289→    [245, 158, 11],
   290→    [249, 115, 22],
   291→    [16, 185, 129],
   292→  ];
   293→
   294→  let curY = y;
   295→
   296→  for (let i = 0; i < phases.length; i++) {
   297→    const p = phases[i];
   298→    const color = colors[i % colors.length];
   299→
   300→    doc.setFont('helvetica', 'normal');
   301→    doc.setFontSize(7);
   302→    doc.setTextColor(50, 50, 50);
   303→    doc.text(p.phase, x, curY + barHeight / 2 + 1, { align: 'right', baseline: 'middle' });
   304→
   305→    doc.setFillColor(...color);
   306→    doc.circle(barStartX - 2, curY + barHeight / 2, 1.5, 'F');
   307→
   308→    const barW = Math.max((p.weeks / totalWeeks) * barMaxWidth, 6);
   309→    doc.setFillColor(color[0], color[1], color[2]);
   310→    doc.setGState(new (doc as any).GState({ opacity: 0.85 }));
   311→    doc.roundedRect(barStartX, curY, barW, barHeight, 1.5, 1.5, 'F');
   312→    doc.setGState(new (doc as any).GState({ opacity: 1 }));
   313→
   314→    doc.setFont('helvetica', 'normal');
   315→    doc.setFontSize(7);
   316→    doc.setTextColor(100, 100, 100);
   317→    const weekLabel = `${p.weeks} week${p.weeks !== 1 ? 's' : ''}`;
   318→    doc.text(weekLabel, barStartX + barW + 3, curY + barHeight / 2 + 1, { align: 'left', baseline: 'middle' });
   319→
   320→    curY += barHeight + gap;
   321→  }
   322→
   323→  return curY;
   324→}

   361→  // ══════════════════════════════════════════════════════════════════════════

   464→    ['IIoT', config.iiot.ipcRequired ? config.iiot.ipcModel : 'Not configured'],
   465→  ];
   466→
   467→  autoTable(doc, {
   468→    startY: y,
   469→    body: techRows,

   675→// ===== POST Handler =====
   676→
   677→export async function POST(request: NextRequest) {
   678→  try {
   679→    const body: ProjectConfig = await request.json();
   680→    const projectId = request.nextUrl.searchParams.get('projectId');
   681→
   682→    const buffer = generatePdfBuffer(body, projectId);
   683→
   684→    const fileName = `br-estimate-${(body.project.name || 'untitled').replace(/[^a-zA-Z0-9_-]/g, '_').toLowerCase()}.pdf`;
   685→
   686→    return new NextResponse(buffer, {
   687→      status: 200,
   688→      headers: {
   689→        'Content-Type': 'application/pdf',
   690→        'Content-Disposition': `attachment; filename="${fileName}"`,
   691→        'Content-Length': String(buffer.byteLength),
   692→      },
   693→    });
   694→  } catch (error) {
   695→    console.error('[PDF Export] Error generating PDF file:', error);
   696→    return NextResponse.json(
   697→      { error: 'Failed to generate PDF file' },
   698→      { status: 500 },
   699→    );
   700→  }
   701→}
