    78→    78→function SortableStatCardWrapper({ id, children, onDragStart }: { id: string; children: React.ReactNode; onDragStart: () => void }) {
    79→    79→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id });

    81→    81→    transform: CSS.Transform.toString(transform),

    83→    83→    opacity: isDragging ? 0.4 : 1,
    84→    84→  };
    85→    85→  return (
    86→    86→    <motion.div
    87→    87→      ref={setNodeRef}
    88→    88→      style={style}
    89→    89→      layout
    90→    90→      className="group-hover:shadow-md transition-shadow duration-300 group relative"
    91→    91→    >
    92→    92→      <div
    93→    93→        {...attributes}
    94→    94→        {...listeners}
    95→    95→        onClick={(e) => e.stopPropagation()}
    96→    96→        onPointerDown={onDragStart}
    97→    97→        className="absolute top-2 right-2 z-10 opacity-0 group-hover:opacity-100 transition-opacity cursor-grab active:cursor-grabbing"
