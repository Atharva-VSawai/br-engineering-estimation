    79→function SortableStatCardWrapper({ id, children, onDragStart }: { id: string; children: React.ReactNode; onDragStart: () => void }) {
    80→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id });
    81→  const style = {
    82→    transform: CSS.Transform.toString(transform),
    83→    transition,
    84→    opacity: isDragging ? 0.4 : 1,
    85→  };
    86→  return (
    87→    <motion.div
    88→      ref={setNodeRef}
    89→      style={style}
    90→      layout
    91→      className="group-hover:shadow-md transition-shadow duration-300 group relative"
    92→    >
    93→      <div
    94→        {...attributes}
    95→        {...listeners}
    96→        onClick={(e) => e.stopPropagation()}
    97→        onPointerDown={onDragStart}
    98→        className="absolute top-2 right-2 z-10 opacity-0 group-hover:opacity-100 transition-opacity cursor-grab active:cursor-grabbing"
    99→      >
   100→        <GripVertical className="h-4 w-4 text-muted-foreground" />
   101→      </div>
   102→      {children}
   103→    </motion.div>
   104→  );
   105→}
