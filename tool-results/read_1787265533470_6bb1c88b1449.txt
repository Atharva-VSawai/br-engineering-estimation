    66→function checkCompleteness(config: ProjectConfig): SectionCheck[] {
    67→  const ioTotal =
    68→    config.io.digitalInputs + config.io.digitalOutputs +
    69→    config.io.analogInputs + config.io.analogOutputs +
    70→    config.io.safetyIO + config.io.encoderCounterModules +
    71→    config.io.temperatureModules + config.io.communicationIO +
    72→    config.io.specialModules;
    73→
    74→  const motionFeatures = [
    75→    config.motion.homingRequired, config.motion.positioning,
    76→    config.motion.velocityControl, config.motion.torqueControl,
    77→    config.motion.synchronization, config.motion.masterSlave,
    78→    config.motion.electronicGearing, config.motion.electronicCamming,
    79→    config.motion.coordinatedMotion, config.motion.interpolation,
    80→    config.motion.complexMotionProfiles, config.motion.axisDiagnostics,
    81→  ].filter(Boolean).length;
    82→
    83→  const hmiFeatures = [
    84→    config.hmi.alarmManagement, config.hmi.recipeManagement,
    85→    config.hmi.trendVisualization, config.hmi.userManagement,
    86→    config.hmi.machineDiagnostics, config.hmi.manualMode,
    87→    config.hmi.automaticMode, config.hmi.maintenanceScreens,
    88→    config.hmi.parameterManagement,
    89→  ].filter(Boolean).length;
    90→
    91→  const commActive = config.communication.protocols.some((p) => p.enabled);
    92→  const commIntegrations = [
    93→    config.communication.plcToPlc, config.communication.mesIntegration,
    94→    config.communication.scadaIntegration, config.communication.cloudIIoTIntegration,
    95→  ].filter(Boolean).length;
    96→
    97→  const iiotFeatures = [
    98→    config.iiot.ipcRequired, config.iiot.iiotRequired,
    99→    config.iiot.iiotConnector, config.iiot.iiotServices,
   100→    config.iiot.iiotEdgeDevice, config.iiot.cloudConnectivity,
   101→    config.iiot.machineDataCollection, config.iiot.remoteMaintenance,
   102→    config.iiot.opcUa, config.iiot.dataLogging, config.iiot.analyticsIntegration,
   103→  ].filter(Boolean).length;
   104→
   105→  return [
   106→    {
   107→      label: 'Project',
   108→      configured: !!(config.project.name && config.project.name.trim()),
   109→    },
   110→    {
   111→      label: 'Controller',
   112→      configured: config.controller.quantity > 1 || config.controller.performance !== 'Standard' || !!config.controller.communicationInterfaces,
   113→    },
   114→    {
   115→      label: 'I/O',
   116→      configured: ioTotal > 0,
   117→    },
   118→    {
   119→      label: 'Motion',
   120→      configured: config.motion.totalAxes > 0 || motionFeatures > 0,
   121→    },
   122→    {
   123→      label: 'HMI',
   124→      configured: config.hmi.screens > 0 || hmiFeatures > 0,
   125→    },
   126→    {
   127→      label: 'Vision',
   128→      configured: config.vision.enabled,
   129→    },
   130→    {
   131→      label: 'Safety',
   132→      configured: config.safety.enabled,
   133→    },
   134→    {
   135→      label: 'Communication',
   136→      configured: commActive || commIntegrations > 0,
   137→    },
   138→    {
   139→      label: 'Mechatronics',
   140→      configured: config.mechatronics.type !== 'None' && config.mechatronics.type !== '',
   141→    },
   142→    {
   143→      label: 'Robotics',
   144→      configured: config.robotics.enabled,
   145→    },
   146→    {
   147→      label: 'IIoT',
   148→      configured: iiotFeatures > 0,
   149→    },
   150→  ];
   151→}
