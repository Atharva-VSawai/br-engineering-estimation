    91→    91→function checkCompleteness(config: ProjectConfig): SectionCheck[] {
    92→    92→  const ioTotal = config.io.digitalInputs + config.io.digitalOutputs + config.io.analogInputs + config.io.analogOutputs + config.io.safetyIO + config.io.encoderCounterModules + config.io.temperatureModules + config.io.communicationIO + config.io.specialModules;
    93→    93→  const motionFeatures = [config.motion.homingRequired, config.motion.positioning, config.motion.velocityControl, config.motion.torqueControl, config.motion.synchronization, config.motion.masterSlave, config.motion.electronicGearing, config.motion.electronicCamming, config.motion.coordinatedMotion, config.motion.interpolation, config.motion.complexMotionProfiles, config.motion.axisDiagnostics].filter(Boolean).length;
    94→    94→  const hmiFeatures = [config.hmi.alarmManagement, config.hmi.recipeManagement, config.hmi.trendVisualization, config.hmi.userManagement, config.hmi.machineDiagnostics, config.hmi.manualMode, config.hmi.automaticMode, config.hmi.maintenanceScreens, config.hmi.parameterManagement].filter(Boolean).length;
    95→    95→  const commActive = config.communication.protocols.some((p) => p.enabled);
    96→    96→  const commIntegrations = [config.communication.plcToPlc, config.communication.mesIntegration, config.communication.scadaIntegration, config.communication.cloudIIoTIntegration].filter(Boolean).length;
    97→    97→  const iiotFeatures = [config.iiot.ipcRequired, config.iiot.iiotRequired, config.iiot.iiotConnector, config.iiot.iiotServices, config.iiot.iiotEdgeDevice, config.iiot.cloudConnectivity, config.iiot.machineDataCollection, config.iiot.remoteMaintenance, config.iiot.opcUa, config.iiot.dataLogging, config.iiot.analyticsIntegration].filter(Boolean).length;
    98→    98→  return [
    99→    99→    { label: 'Project', configured: !!(config.project.name && config.project.name.trim()) },
   100→   100→    { label: 'Controller', configured: config.controller.quantity > 1 || config.controller.performance !== 'Standard' || !!config.controller.communicationInterfaces },
   101→   101→    { label: 'I/O', configured: ioTotal > 0 },
   102→   102→    { label: 'Motion', configured: config.motion.totalAxes > 0 || motionFeatures > 0 },
   103→   103→    { label: 'HMI', configured: config.hmi.screens > 0 || hmiFeatures > 0 },
   104→   104→    { label: 'Vision', configured: config.vision.enabled },
   105→   105→    { label: 'Safety', configured: config.safety.enabled },
   106→   106→    { label: 'Communication', configured: commActive || commIntegrations > 0 },
   107→   107→    { label: 'Mechatronics', configured: config.mechatronics.type !== 'None' && config.mechatronics.type !== '' },
   108→   108→    { label: 'Robotics', configured: config.robotics.enabled },
   109→   109→    { label: 'IIoT', configured: iiotFeatures > 0 },
   110→   110→  ];
   111→   111→}
   112→   112→
   113→   113→function getEffortDrivers(c: ProjectConfig): string[] {
   114→   114→  const drivers: string[] = [];
   115→   115→  if (c.motion.totalAxes > 0) drivers.push(`${c.motion.totalAxes} motion axes`);
   116→   116→  if (c.motion.electronicCamming) drivers.push('Electronic camming');
   117→   117→  if (c.motion.coordinatedMotion) drivers.push('Coordinated motion');
   118→   118→  if (c.motion.interpolation) drivers.push('Multi-axis interpolation');
   119→   119→  if (c.motion.complexMotionProfiles) drivers.push('Complex motion profiles');
   120→   120→  if (c.mechatronics.type !== 'None') drivers.push(`${c.mechatronics.type} mechatronics`);
   121→   121→  if (c.robotics.enabled) drivers.push(`${c.robotics.quantity} robot(s)`);
   122→   122→  if (c.vision.enabled) drivers.push(`${c.vision.cameras} camera(s)`);
   123→   123→  if (c.safety.enabled) drivers.push(`Safety system (${c.safety.safetyIOCount} I/O)`);
   124→   124→  if (c.hmi.screens > 0) drivers.push(`${c.hmi.screens} HMI screen(s)`);
   125→   125→  const enabledProtocols = c.communication.protocols.filter((p) => p.enabled);
   126→   126→  if (enabledProtocols.length > 0) drivers.push(`${enabledProtocols.length} communication protocol(s)`);
   127→   127→  if (c.communication.mesIntegration) drivers.push('MES integration');
   128→   128→  if (c.communication.scadaIntegration) drivers.push('SCADA integration');
   129→   129→  if (c.iiot.ipcRequired) drivers.push(`IPC (${c.iiot.ipcModel})`);
   130→   130→  if (c.iiot.cloudConnectivity) drivers.push('Cloud/IIoT connectivity');
   131→   131→  const totalIO = c.io.digitalInputs + c.io.digitalOutputs + c.io.analogInputs + c.io.analogOutputs;
   132→   132→  if (totalIO > 100) drivers.push(`${totalIO} I/O points`);
   133→   133→  if (c.project.machineStations > 1) drivers.push(`${c.project.machineStations} machine stations`);
   134→   134→  if (c.project.projectVariants > 1) drivers.push(`${c.project.projectVariants} product variants`);
   135→   135→  return drivers;
   136→   136→}
