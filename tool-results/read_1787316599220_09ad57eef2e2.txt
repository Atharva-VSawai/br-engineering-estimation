   129→function checkCompleteness(config: ProjectConfig): SectionCheck[] {
   130→  const ioTotal = config.io.digitalInputs + config.io.digitalOutputs + config.io.analogInputs + config.io.analogOutputs + config.io.safetyIO + config.io.encoderCounterModules + config.io.temperatureModules + config.io.communicationIO + config.io.specialModules;
   131→  const motionFeatures = [config.motion.homingRequired, config.motion.positioning, config.motion.velocityControl, config.motion.torqueControl, config.motion.synchronization, config.motion.masterSlave, config.motion.electronicGearing, config.motion.electronicCamming, config.motion.coordinatedMotion, config.motion.interpolation, config.motion.complexMotionProfiles, config.motion.axisDiagnostics].filter(Boolean).length;
   132→  const hmiFeatures = [config.hmi.alarmManagement, config.hmi.recipeManagement, config.hmi.trendVisualization, config.hmi.userManagement, config.hmi.machineDiagnostics, config.hmi.manualMode, config.hmi.automaticMode, config.hmi.maintenanceScreens, config.hmi.parameterManagement].filter(Boolean).length;
   133→  const commActive = config.communication.protocols.some((p) => p.enabled);
   134→  const commIntegrations = [config.communication.plcToPlc, config.communication.mesIntegration, config.communication.scadaIntegration, config.communication.cloudIIoTIntegration].filter(Boolean).length;
   135→  const iiotFeatures = [config.iiot.ipcRequired, config.iiot.iiotRequired, config.iiot.iiotConnector, config.iiot.iiotServices, config.iiot.iiotEdgeDevice, config.iiot.cloudConnectivity, config.iiot.machineDataCollection, config.iiot.remoteMaintenance, config.iiot.opcUa, config.iiot.dataLogging, config.iiot.analyticsIntegration].filter(Boolean).length;
   136→  return [
   137→    { label: 'Project', configured: !!(config.project.name && config.project.name.trim()) },
   138→    { label: 'Controller', configured: config.controller.quantity > 1 || config.controller.performance !== 'Standard' || !!config.controller.communicationInterfaces },
   139→    { label: 'I/O', configured: ioTotal > 0 },
   140→    { label: 'Motion', configured: config.motion.totalAxes > 0 || motionFeatures > 0 },
   141→    { label: 'HMI', configured: config.hmi.screens > 0 || hmiFeatures > 0 },
   142→    { label: 'Vision', configured: config.vision.enabled },
   143→    { label: 'Safety', configured: config.safety.enabled },
   144→    { label: 'Communication', configured: commActive || commIntegrations > 0 },
   145→    { label: 'Mechatronics', configured: config.mechatronics.type !== 'None' && config.mechatronics.type !== '' },
   146→    { label: 'Robotics', configured: config.robotics.enabled },
   147→    { label: 'IIoT', configured: iiotFeatures > 0 },
   148→  ];
   149→}

   151→function getEffortDrivers(c: ProjectConfig): string[] {
   152→  const drivers: string[] = [];
   153→  if (c.motion.totalAxes > 0) drivers.push(`${c.motion.totalAxes} motion axes`);
   154→  if (c.motion.electronicCamming) drivers.push('Electronic camming');
   155→  if (c.motion.coordinatedMotion) drivers.push('Coordinated motion');
   156→  if (c.motion.interpolation) drivers.push('Multi-axis interpolation');
   157→  if (c.motion.complexMotionProfiles) drivers.push('Complex motion profiles');
   158→  if (c.mechatronics.type !== 'None') drivers.push(`${c.mechatronics.type} mechatronics`);
   159→  if (c.robotics.enabled) drivers.push(`${c.robotics.quantity} robot(s)`);
   160→  if (c.vision.enabled) drivers.push(`${c.vision.cameras} camera(s)`);
   161→  if (c.safety.enabled) drivers.push(`Safety system (${c.safety.safetyIOCount} I/O)`);
   162→  if (c.hmi.screens > 0) drivers.push(`${c.hmi.screens} HMI screen(s)`);
   163→  const enabledProtocols = c.communication.protocols.filter((p) => p.enabled);
   164→  if (enabledProtocols.length > 0) drivers.push(`${enabledProtocols.length} communication protocol(s)`);
   165→  if (c.communication.mesIntegration) drivers.push('MES integration');
   166→  if (c.communication.scadaIntegration) drivers.push('SCADA integration');
   167→  if (c.iiot.ipcRequired) drivers.push(`IPC (${c.iiot.ipcModel})`);
   168→  if (c.iiot.cloudConnectivity) drivers.push('Cloud/IIoT connectivity');
   169→  const totalIO = c.io.digitalInputs + c.io.digitalOutputs + c.io.analogInputs + c.io.analogOutputs;
   170→  if (totalIO > 100) drivers.push(`${totalIO} I/O points`);
   171→  if (c.project.machineStations > 1) drivers.push(`${c.project.machineStations} machine stations`);
   172→  if (c.project.projectVariants > 1) drivers.push(`${c.project.projectVariants} product variants`);
   173→  return drivers;
   174→}

   190→export function EstimateSummaryPage() {
   191→  const { config: c, activeProjectId, projects, setCurrentPage, setWizardStep, addProject, openProject } = useAppStore();

   193→

   195→  const [viewMode, setViewMode] = useState<'all' | string>('all');
   196→
   197→  // Calculate results for ALL projects with configs
   198→  const projectResults = useMemo(() => {
   199→    return projects
   200→      .filter((p) => p.config)
   201→      .map((p) => {
   202→        const result = calculateEngineeringEffort(p.config!);
   203→        return { project: p, ...result };
   204→      });
   205→  }, [projects]);
   206→
   207→  // Combined (aggregated) effort from ALL projects - dynamically calculated
   208→  const combinedEffort = useMemo(() => {
   209→    if (projectResults.length === 0) {
   210→      // Fallback: use current working config
   211→      const result = calculateEngineeringEffort(c);
   212→      return result.effort;
   213→    }
   214→    const base: EffortResult = {
   215→      hardwareHours: 0, plcSoftwareHours: 0, motionHours: 0,
   216→      hmiHours: 0, visionHours: 0, safetyHours: 0,
   217→      communicationIntegrationHours: 0, testingHours: 0, commissioningHours: 0,
   218→      totalHours: 0, totalDays: 0, totalWeeks: 0, totalMonths: 0,
   219→    };
   220→    for (const pr of projectResults) {
   221→      base.hardwareHours += pr.effort.hardwareHours;
   222→      base.plcSoftwareHours += pr.effort.plcSoftwareHours;
   223→      base.motionHours += pr.effort.motionHours;
   224→      base.hmiHours += pr.effort.hmiHours;
   225→      base.visionHours += pr.effort.visionHours;
   226→      base.safetyHours += pr.effort.safetyHours;
   227→      base.communicationIntegrationHours += pr.effort.communicationIntegrationHours;
   228→      base.testingHours += pr.effort.testingHours;
   229→      base.commissioningHours += pr.effort.commissioningHours;
   230→    }
   231→    base.totalHours = base.hardwareHours + base.plcSoftwareHours + base.motionHours + base.hmiHours + base.visionHours + base.safetyHours + base.communicationIntegrationHours + base.testingHours + base.commissioningHours;
   232→    base.totalDays = base.totalHours / 8;
   233→    base.totalWeeks = Math.ceil(base.totalDays / 5);
   234→    base.totalMonths = Math.round((base.totalDays / 20) * 10) / 10;
   235→    return base;
   236→  }, [projectResults, c]);
   237→
   238→  // Combined timeline from ALL projects
   239→  const combinedTimeline = useMemo(() => {
   240→    if (projectResults.length === 0) {
   241→      const result = calculateEngineeringEffort(c);
   242→      return result.timeline;
   243→    }
   244→    const base: TimelineResult = { hardwareDesignWeeks: 0, softwareDevelopmentWeeks: 0, integrationTestingWeeks: 0, commissioningWeeks: 0, totalWeeks: 0 };
   245→    for (const pr of projectResults) {
   246→      base.hardwareDesignWeeks += pr.timeline.hardwareDesignWeeks;
   247→      base.softwareDevelopmentWeeks += pr.timeline.softwareDevelopmentWeeks;
   248→      base.integrationTestingWeeks += pr.timeline.integrationTestingWeeks;
   249→      base.commissioningWeeks += pr.timeline.commissioningWeeks;
   250→    }
   251→    base.totalWeeks = base.hardwareDesignWeeks + base.softwareDevelopmentWeeks + base.integrationTestingWeeks + base.commissioningWeeks;
   252→    return base;
   253→  }, [projectResults, c]);
   254→
   255→  // Determine which data to display based on viewMode
   256→  const displayConfig = useMemo((): ProjectConfig => {
   257→    if (viewMode === 'all') return c;
   258→    const p = projects.find((pr) => pr.id === viewMode);
   259→    if (p?.config) return p.config;
   260→    return c;
   261→  }, [viewMode, projects, c]);
   262→
   263→  const displayResult = useMemo(() => {
   264→    if (viewMode === 'all') {
   265→      // For combined view, we still need overallComplexity and highCount from displayConfig
   266→      return calculateEngineeringEffort(displayConfig);
   267→    }
   268→    return calculateEngineeringEffort(displayConfig);
   269→  }, [viewMode, displayConfig]);
   270→
   271→  // The actual effort data to render (combined or per-project)
   272→  const effort = viewMode === 'all' ? combinedEffort : displayResult.effort;
   273→  const timeline = viewMode === 'all' ? combinedTimeline : displayResult.timeline;
   274→  const overallComplexity = displayResult.overallComplexity;
   275→  const highCount = displayResult.highCount;
   276→
   277→  const areaComplexities: Record<string, ComplexityLevel> = {
   278→    Motion: displayConfig.complexity.motion,
   279→    HMI: displayConfig.complexity.hmi,
   280→    'I/O': displayConfig.io.digitalInputs + displayConfig.io.digitalOutputs > 200 ? 'High' : 'Medium',
   281→    Vision: displayConfig.vision.enabled ? displayConfig.complexity.vision : 'Low',
   282→    Safety: displayConfig.safety.enabled ? displayConfig.complexity.safety : 'Low',
   283→    Communication: displayConfig.complexity.communication,
   284→    Software: displayConfig.complexity.software,

   289→

   292→  const drivers = useMemo(() => getEffortDrivers(displayConfig), [displayConfig]);
   293→
   294→  // Donut chart data
   295→  const effortChartData = EFFORT_KEYS.map(({ key, label }) => ({
   296→    name: label,
   297→    value: effort[key],
   298→    color: EFFORT_CHART_COLORS[label],

   300→  const chartData = effortChartData.filter((d) => d.value > 0);
