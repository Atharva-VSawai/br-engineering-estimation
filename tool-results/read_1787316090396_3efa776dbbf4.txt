    79→    79→function SortableStatCardWrapper({ id, children, onDragStart }: { id: string; children: React.ReactNode; onDragStart: () => void }) {

    87→    87→    <motion.div
    88→    88→      ref={setNodeRef}
    89→    89→      style={style}
    90→    90→      layout
    91→    91→      className="group-hover:shadow-md transition-shadow duration-300 group relative"
    92→    92→    >
    93→    93→      <div
    94→    94→        {...attributes}
    95→    95→        {...listeners}
    96→    96→        onClick={(e) => e.stopPropagation()}
    97→    97→        onPointerDown={onDragStart}
