    78→function SortableStatCardWrapper({ id, children, onDragStart }: { id: string; children: React.ReactNode; onDragStart: () => void }) {
    79→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id });
    80→  const style = {
    81→    transform: CSS.Transform.toString(transform),
    82→    transition,
    83→    opacity: isDragging ? 0.4 : 1,
    84→  };
    85→  return (
    86→    <motion.div
    87→      ref={setNodeRef}
    88→      style={style}
    89→      layout
    90→      className="group-hover:shadow-md transition-shadow duration-300 group relative"
    91→    >
    92→      <div
    93→        {...attributes}
    94→        {...listeners}
    95→        onClick={(e) => e.stopPropagation()}
    96→        onPointerDown={onDragStart}
    97→        className="absolute top-2 right-2 z-10 opacity-0 group-hover:opacity-100 transition-opacity cursor-grab active:cursor-grabbing"
    98→      >
    99→        <GripVertical className="h-4 w-4 text-muted-foreground" />
   100→      </div>
   101→      {children}
   102→    </motion.div>
   103→  );
   104→}
