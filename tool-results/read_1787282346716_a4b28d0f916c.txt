    71→    71→function SortableStatCardWrapper({ id, children, onDragStart }: { id: string; children: React.ReactNode; onDragStart: () => void }) {
    72→    72→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id });
    73→    73→  const style = {
    74→    74→    transform: CSS.Transform.toString(transform),
    75→    75→    transition,
    76→    76→    opacity: isDragging ? 0.4 : 1,
    77→    77→  };
    78→    78→  return (
    79→    79→    <motion.div
    80→    80→      ref={setNodeRef}
    81→    81→      style={style}
    82→    82→      layout
    83→    83→      className="group-hover:shadow-md transition-shadow duration-300 group relative"
    84→    84→    >
    85→    85→      <div
    86→    86→        {...attributes}
    87→    87→        {...listeners}
    88→    88→        onClick={(e) => e.stopPropagation()}
    89→    89→        onPointerDown={onDragStart}
    90→    90→        className="absolute top-2 right-2 z-10 opacity-0 group-hover:opacity-100 transition-opacity cursor-grab active:cursor-grabbing"
    91→    91→      >
    92→    92→        <GripVertical className="h-4 w-4 text-muted-foreground" />
    93→    93→      </div>
    94→    94→      {children}
    95→    95→    </motion.div>
    96→    96→  );
    97→    97→}
