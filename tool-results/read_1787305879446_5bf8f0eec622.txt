    53→    53→function getOverallComplexity(config: ProjectConfig): ComplexityLevel {
    54→    54→  const dims = DIMENSIONS.map((d) => config.complexity[d.key]);
    55→    55→  const highCount = dims.filter((x) => x === 'High' || x === 'Very High').length;
    56→    56→  if (highCount >= 5) return 'Very High';
    57→    57→  if (highCount >= 3) return 'High';
    58→    58→  if (highCount >= 1) return 'Medium';
    59→    59→  return 'Low';
    60→    60→}

    62→    62→function calcEffort(config: ProjectConfig) {
    63→    63→  const c = config;
    64→    64→  const hwHours = (c.io.digitalInputs + c.io.digitalOutputs + c.io.analogInputs + c.io.analogOutputs) * 0.5 + c.motion.totalAxes * 4;
    65→    65→  const swHours = c.hmi.screens * 8 + (c.vision.enabled ? c.vision.cameras * 16 : 0) + 40;
    66→    66→  const motionHours = c.motion.totalAxes * 6 + (c.motion.electronicCamming ? 20 : 0) + (c.motion.coordinatedMotion ? 16 : 0);
    67→    67→  const safetyHours = c.safety.enabled ? c.safety.safetyIOCount * 2 + 16 : 0;
    68→    68→  const integrationHours = (hwHours + swHours + motionHours + safetyHours) * 0.3;
    69→    69→  const totalHours = hwHours + swHours + motionHours + safetyHours + integrationHours;
    70→    70→  return { hwHours, swHours, motionHours, safetyHours, integrationHours, totalHours };
    71→    71→}

   164→   164→function drawRadarChart(doc: jsPDF, cx: number, cy: number, maxR: number, config: ProjectConfig): void {
   165→   165→  const n = DIMENSIONS.length;
   166→   166→  const angleStep = (2 * Math.PI) / n;
   167→   167→
   168→   168→  // Draw concentric rings (25%, 50%, 75%, 100%)
   169→   169→  for (const ring of [0.25, 0.5, 0.75, 1.0]) {
   170→   170→    const r = ring * maxR;
   171→   171→    const pts: [number, number][] = [];
   172→   172→    for (let i = 0; i < n; i++) {
   173→   173→      const angle = i * angleStep - Math.PI / 2;
   174→   174→      pts.push([cx + r * Math.cos(angle), cy + r * Math.sin(angle)]);
   175→   175→    }
   176→   176→    doc.setDrawColor(220, 220, 220);
   177→   177→    doc.setLineWidth(0.3);
   178→   178→    drawPolygonStroke(doc, pts);
   179→   179→  }
   180→   180→
   181→   181→  // Draw axis lines
   182→   182→  for (let i = 0; i < n; i++) {
   183→   183→    const angle = i * angleStep - Math.PI / 2;
   184→   184→    const px = cx + maxR * Math.cos(angle);
   185→   185→    const py = cy + maxR * Math.sin(angle);
   186→   186→    doc.setDrawColor(220, 220, 220);
   187→   187→    doc.setLineWidth(0.3);
   188→   188→    doc.line(cx, cy, px, py);
   189→   189→  }
   190→   190→
   191→   191→  // Draw data polygon
   192→   192→  const dataPts: [number, number][] = [];
   193→   193→  for (let i = 0; i < n; i++) {
   194→   194→    const dim = DIMENSIONS[i];
   195→   195→    const val = COMPLEXITY_VALUE[config.complexity[dim.key]] || 0.25;
   196→   196→    const r = val * maxR;
   197→   197→    const angle = i * angleStep - Math.PI / 2;
   198→   198→    dataPts.push([cx + r * Math.cos(angle), cy + r * Math.sin(angle)]);
   199→   199→  }
   200→   200→
   201→   201→  // Filled polygon with semi-transparent orange
   202→   202→  doc.setFillColor(199, 91, 18);
   203→   203→  doc.setGState(new (doc as any).GState({ opacity: 0.2 }));
   204→   204→  drawPolygonFill(doc, dataPts);
   205→   205→  doc.setGState(new (doc as any).GState({ opacity: 1 }));
   206→   206→  // Stroke polygon
   207→   207→  doc.setDrawColor(...BR_ORANGE);
   208→   208→  doc.setLineWidth(1.2);
   209→   209→  drawPolygonStroke(doc, dataPts);
   210→   210→
   211→   211→  // Labels
   212→   212→  doc.setFont('helvetica', 'normal');
   213→   213→  doc.setFontSize(7);
   214→   214→  doc.setTextColor(80, 80, 80);
   215→   215→  for (let i = 0; i < n; i++) {
   216→   216→    const angle = i * angleStep - Math.PI / 2;
   217→   217→    const lx = cx + (maxR + 5) * Math.cos(angle);
   218→   218→    const ly = cy + (maxR + 5) * Math.sin(angle);
   219→   219→    const label = DIMENSIONS[i].label;
   220→   220→    if (Math.abs(Math.cos(angle)) < 0.15) {
   221→   221→      doc.text(label, lx, ly, { align: 'center', baseline: 'middle' });
   222→   222→    } else if (Math.cos(angle) > 0) {
   223→   223→      doc.text(label, lx, ly, { align: 'left', baseline: 'middle' });
   224→   224→    } else {
   225→   225→      doc.text(label, lx, ly, { align: 'right', baseline: 'middle' });
   226→   226→    }
   227→   227→  }
   228→   228→}

   230→   230→// ── Horizontal Bar Chart ─────────────────────────────────────────────────────
   231→   231→
   232→   232→function drawHorizontalBarChart(
   233→   233→  doc: jsPDF,
   234→   234→  x: number,
   235→   235→  y: number,
   236→   236→  width: number,
   237→   237→  barHeight: number,
   238→   238→  gap: number,
   239→   239→  items: { label: string; value: number; max: number; color: [number, number, number]; valueLabel?: string }[],
   240→   240→  showTotal?: { label: string; value: number; color: [number, number, number] },
   241→   241→): number {
   242→   242→  const labelWidth = 38;
   243→   243→  const barStartX = x + labelWidth + 4;
   244→   244→  const barMaxWidth = width - labelWidth - 30;
   245→   245→  const maxVal = Math.max(...items.map((it) => it.max), 1);
   246→   246→
   247→   247→  let curY = y;
   248→   248→
   249→   249→  for (const item of items) {
   250→   250→    // Label
   251→   251→    doc.setFont('helvetica', 'normal');
   252→   252→    doc.setFontSize(7);
   253→   253→    doc.setTextColor(50, 50, 50);
   254→   254→    doc.text(item.label, x, curY + barHeight / 2 + 1, { align: 'right', baseline: 'middle' });
   255→   255→
   256→   256→    // Bar
   257→   257→    const barW = Math.max((item.value / maxVal) * barMaxWidth, 2);
   258→   258→    doc.setFillColor(...item.color);
   259→   259→    doc.roundedRect(barStartX, curY, barW, barHeight, 1.5, 1.5, 'F');
   260→   260→
   261→   261→    // Value label
   262→   262→    doc.setFont('helvetica', 'bold');
   263→   263→    doc.setFontSize(7);
   264→   264→    doc.setTextColor(50, 50, 50);
   265→   265→    const vLabel = item.valueLabel ?? `${item.value.toFixed(1)}h`;
   266→   266→    doc.text(vLabel, barStartX + barW + 3, curY + barHeight / 2 + 1, { align: 'left', baseline: 'middle' });
   267→   267→
   268→   268→    curY += barHeight + gap;
   269→   269→  }
   270→   270→
   271→   271→  // Optional total row
   272→   272→  if (showTotal) {
   273→   273→    // Separator line
   274→   274→    curY -= gap / 2;
   275→   275→    doc.setDrawColor(220, 220, 220);
   276→   276→    doc.setLineWidth(0.3);
   277→   277→    doc.line(barStartX, curY, barStartX + barMaxWidth, curY);
   278→   278→    curY += gap / 2;
   279→   279→
   280→   280→    // Label
   281→   281→    doc.setFont('helvetica', 'bold');
   282→   282→    doc.setFontSize(7);
   283→   283→    doc.setTextColor(...BR_ORANGE);
   284→   284→    doc.text(showTotal.label, x, curY + barHeight / 2 + 1, { align: 'right', baseline: 'middle' });
   285→   285→
   286→   286→    // Bar
   287→   287→    const barW = Math.max((showTotal.value / maxVal) * barMaxWidth, 2);
   288→   288→    doc.setFillColor(...showTotal.color);
   289→   289→    doc.roundedRect(barStartX, curY, barW, barHeight, 1.5, 1.5, 'F');
   290→   290→
   291→   291→    // Value
   292→   292→    doc.text(`${showTotal.value.toFixed(1)}h`, barStartX + barW + 3, curY + barHeight / 2 + 1, { align: 'left', baseline: 'middle' });
   293→   293→    curY += barHeight + gap;
   294→   294→  }
   295→   295→
   296→   296→  return curY;
   297→   297→}

   299→   299→// ── Timeline / Gantt Chart ────────────────────────────────────────────────────
   300→   300→
   301→   301→function drawTimelineChart(
   302→   302→  doc: jsPDF,
   303→   303→  x: number,
   304→   304→  y: number,
   305→   305→  width: number,
   306→   306→  barHeight: number,
   307→   307→  gap: number,
   308→   308→  phases: { phase: string; weeks: number }[],
   309→   309→  totalWeeks: number,
   310→   310→): number {
   311→   311→  const labelWidth = 40;
   312→   312→  const barStartX = x + labelWidth + 4;
   313→   313→  const barMaxWidth = width - labelWidth - 28;
   314→   314→  const colors: [number, number, number][] = [
   315→   315→    [59, 130, 246],
   316→   316→    [245, 158, 11],
   317→   317→    [249, 115, 22],
   318→   318→    [16, 185, 129],
   319→   319→  ];
   320→   320→
   321→   321→  let curY = y;
   322→   322→
   323→   323→  for (let i = 0; i < phases.length; i++) {
   324→   324→    const p = phases[i];
   325→   325→    const color = colors[i % colors.length];
   326→   326→
   327→   327→    // Label
   328→   328→    doc.setFont('helvetica', 'normal');
   329→   329→    doc.setFontSize(7);
   330→   330→    doc.setTextColor(50, 50, 50);
   331→   331→    doc.text(p.phase, x, curY + barHeight / 2 + 1, { align: 'right', baseline: 'middle' });
   332→   332→
   333→   333→    // Dot
   334→   334→    doc.setFillColor(...color);
   335→   335→    doc.circle(barStartX - 2, curY + barHeight / 2, 1.5, 'F');
   336→   336→
   337→   337→    // Bar
   338→   338→    const barW = Math.max((p.weeks / totalWeeks) * barMaxWidth, 6);
   339→   339→    doc.setFillColor(color[0], color[1], color[2]);
   340→   340→    doc.setGState(new (doc as any).GState({ opacity: 0.85 }));
   341→   341→    doc.roundedRect(barStartX, curY, barW, barHeight, 1.5, 1.5, 'F');
   342→   342→    doc.setGState(new (doc as any).GState({ opacity: 1 }));
   343→   343→
   344→   344→    // Value
   345→   345→    doc.setFont('helvetica', 'normal');
   346→   346→    doc.setFontSize(7);
   347→   347→    doc.setTextColor(100, 100, 100);
   348→   348→    const weekLabel = `${p.weeks} week${p.weeks !== 1 ? 's' : ''}`;
   349→   349→    doc.text(weekLabel, barStartX + barW + 3, curY + barHeight / 2 + 1, { align: 'left', baseline: 'middle' });
   350→   350→
   351→   351→    curY += barHeight + gap;
   352→   352→  }
   353→   353→
   354→   354→  return curY;
   355→   355→}

   404→   404→// ── Risk Assessment Cards ─────────────────────────────────────────────────────
   405→   405→
   406→   406→function drawRiskAssessment(doc: jsPDF, y: number, config: ProjectConfig, highCount: number): number {
   407→   407→  const clarity = config.project.requirementClarity || 'Mostly Clear';
   408→   408→  const involvement = config.project.customerInvolvement || 'Medium';
   409→   409→
   410→   410→  const clarityColorMap: Record<string, [number, number, number]> = {
   411→   411→    Clear: [16, 185, 129],
   412→   412→    'Mostly Clear': [245, 158, 11],
   413→   413→    'Partially Clear': [249, 115, 22],
   414→   414→    Unclear: [239, 68, 68],
   415→   415→  };
   416→   416→  const involvementColorMap: Record<string, [number, number, number]> = {
   417→   417→    Low: [239, 68, 68],
   418→   418→    Medium: [245, 158, 11],
   419→   419→    High: [16, 185, 129],
   420→   420→  };
   421→   421→  const scopeColor: [number, number, number] =
   422→   422→    highCount >= 5 ? [239, 68, 68] : highCount >= 3 ? [249, 115, 22] : highCount >= 1 ? [245, 158, 11] : [16, 185, 129];
   423→   423→
   424→   424→  const cardW = 55;
   425→   425→  const cardH = 22;
   426→   426→  const gap = 5;
   427→   427→  const startX = 15;
   428→   428→
   429→   429→  const cards = [
   430→   430→    { label: 'Requirement Clarity', value: clarity, color: clarityColorMap[clarity] || [150, 150, 150] },
   431→   431→    { label: 'Customer Involvement', value: involvement, color: involvementColorMap[involvement] || [150, 150, 150] },
   432→   432→    { label: 'Scope Complexity', value: `${highCount} / 10 dimensions`, color: scopeColor },
   433→   433→  ];
   434→   434→
   435→   435→  for (let i = 0; i < cards.length; i++) {
   436→   436→    const cx = startX + i * (cardW + gap);
   437→   437→    const card = cards[i];
   438→   438→
   439→   439→    // Card background
   440→   440→    doc.setFillColor(248, 248, 248);
   441→   441→    doc.setDrawColor(220, 220, 220);
   442→   442→    doc.setLineWidth(0.3);
   443→   443→    doc.roundedRect(cx, y, cardW, cardH, 2, 2, 'FD');
   444→   444→
   445→   445→    // Label
   446→   446→    doc.setFont('helvetica', 'normal');
   447→   447→    doc.setFontSize(6);
   448→   448→    doc.setTextColor(130, 130, 130);
   449→   449→    doc.text(card.label.toUpperCase(), cx + 3, y + 6);
   450→   450→
   451→   451→    // Color dot
   452→   452→    doc.setFillColor(...card.color);
   453→   453→    doc.circle(cx + 5, y + 14, 2, 'F');
   454→   454→
   455→   455→    // Value
   456→   456→    doc.setFont('helvetica', 'bold');
   457→   457→    doc.setFontSize(9);
   458→   458→    doc.setTextColor(30, 30, 30);
   459→   459→    doc.text(card.value, cx + 9, y + 15.5);
   460→   460→  }
   461→   461→
   462→   462→  return y + cardH + 4;
   463→   463→}
